cmake_minimum_required(VERSION 3.16)
project(velaride_sport_ui_test C CXX ASM)
option(VELARIDE_COUNTDOWN "Exercise standalone countdown" OFF)
option(VELARIDE_ALARM "Exercise daily alarm" OFF)
option(VELARIDE_SOUND "Exercise sound settings and event wiring" OFF)
option(VELARIDE_NIGHT "Exercise night display" OFF)
if(VELARIDE_NIGHT)
  add_compile_definitions(LV_USE_COLOR_FILTER=1)
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
if(NOT LVGL_SOURCE)
  message(FATAL_ERROR "Set LVGL_SOURCE to the project LVGL source directory")
endif()
add_subdirectory("${LVGL_SOURCE}" lvgl-build)
set(TITLE_FONT ../../app/velaride/velaride_font_cn_28.c)
if(VELARIDE_COUNTDOWN)
  set(TITLE_FONT ../../app/velaride/velaride_font_countdown_28.c)
endif()
if(VELARIDE_ALARM)
  set(TITLE_FONT ../../app/velaride/velaride_font_alarm_28.c)
endif()
if(VELARIDE_SOUND)
  set(TITLE_FONT ../../app/velaride/velaride_font_sound_28.c)
endif()
if(VELARIDE_NIGHT)
  set(TITLE_FONT ../../app/velaride/velaride_font_night_28.c)
endif()
set(VELARIDE_FONTS ../../app/velaride/velaride_font_cn_20.c ${TITLE_FONT}
  ../../app/velaride/velaride_num_136.c ../../app/velaride/velaride_num_92.c ../../app/velaride/velaride_num_60.c ../../app/velaride/velaride_num_44.c ../../app/velaride/velaride_num_30.c ../../app/velaride/velaride_num_24.c)
add_executable(sport_ui_test test.c ${VELARIDE_FONTS})
get_filename_component(LVGL_PARENT "${LVGL_SOURCE}" DIRECTORY)
target_include_directories(sport_ui_test PRIVATE ../ride_session/stubs "${LVGL_PARENT}")
target_link_libraries(sport_ui_test PRIVATE lvgl m)
target_compile_options(sport_ui_test PRIVATE -Wall -Wextra -Werror)

# Use the actual shell UI, excluding only the NSH/hardware entry helpers.
set(SHELL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../app/velaride/velaride_shell.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SHELL_SOURCE}")
file(READ "${SHELL_SOURCE}" SHELL_CODE)
string(FIND "${SHELL_CODE}" "static bool wait_for_device(" SHELL_CUT)
if(SHELL_CUT LESS 0)
  message(FATAL_ERROR "Shell UI boundary changed; update host test deliberately")
endif()
string(SUBSTRING "${SHELL_CODE}" 0 ${SHELL_CUT} SHELL_UI_CODE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shell-ui.c" "${SHELL_UI_CODE}")
add_executable(shell_ui_test shell_test.c ${VELARIDE_FONTS})
target_include_directories(shell_ui_test PRIVATE ../ride_session/stubs
  ../../app/velaride "${LVGL_PARENT}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(shell_ui_test PRIVATE lvgl m)
target_compile_options(shell_ui_test PRIVATE -Wall -Wextra -Werror
  -Wno-unused-variable -Wno-unused-function)

if(VELARIDE_SOUND)
  foreach(target sport_ui_test shell_ui_test)
    target_compile_definitions(${target} PRIVATE CONFIG_EXAMPLES_VELARIDE_SOUND=1)
    target_sources(${target} PRIVATE sound_fake.c)
    target_include_directories(${target} PRIVATE ../../app/velaride)
  endforeach()
endif()

if(VELARIDE_COUNTDOWN)
  foreach(target sport_ui_test shell_ui_test)
    target_compile_definitions(${target} PRIVATE CONFIG_EXAMPLES_VELARIDE_COUNTDOWN=1)
    target_sources(${target} PRIVATE ../../app/velaride/velaride_countdown.c)
    target_link_libraries(${target} PRIVATE pthread)
  endforeach()
endif()
if(VELARIDE_ALARM)
  foreach(target sport_ui_test shell_ui_test)
    target_compile_definitions(${target} PRIVATE CONFIG_EXAMPLES_VELARIDE_ALARM=1)
    target_sources(${target} PRIVATE ../../app/velaride/velaride_alarm.c)
    target_link_libraries(${target} PRIVATE pthread)
  endforeach()
endif()
if(VELARIDE_ALARM OR VELARIDE_NIGHT)
  foreach(target sport_ui_test shell_ui_test)
    target_sources(${target} PRIVATE alarm_store.c)
  endforeach()
endif()
if(VELARIDE_NIGHT)
  foreach(target sport_ui_test shell_ui_test)
    target_compile_definitions(${target} PRIVATE CONFIG_EXAMPLES_VELARIDE_NIGHT=1)
    target_sources(${target} PRIVATE ../../app/velaride/velaride_night.c
      ../../app/velaride/velaride_night_ui.c)
    target_link_libraries(${target} PRIVATE pthread)
  endforeach()
endif()
